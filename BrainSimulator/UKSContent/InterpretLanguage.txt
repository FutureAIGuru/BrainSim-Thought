# A phrase can supply a semantic Thought and supporting Relationships.
# Sentence classes determine intent. The interpreter returns ordinary types
# after resolving these parameterized MEANS Relationships.

[interpret:noun->hasWords->^slot:noun]
[interpret:noun->means->slot:noun]
[interpret:articleNoun->is-a->NounPhrase]
[interpret:articleNoun->hasWords->^slot:article slot:noun]
[interpret:articleNoun->means->slot:noun]
[interpret:describedNoun->is-a->NounPhrase]
[interpret:describedNoun->hasWords->^slot:adjective slot:noun]
[interpret:describedNoun->means->slot:noun]
[interpret:describedNoun->means->[slot:noun->is.interpretWrite->slot:adjective]]
[interpret:articlePhrase->is-a->NounPhrase]
[interpret:articlePhrase->hasWords->^slot:article slot:object]
[interpret:articlePhrase->means->slot:object]
