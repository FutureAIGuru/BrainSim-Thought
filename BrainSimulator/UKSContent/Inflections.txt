# Inflection rules (ModuleInflect). A regular word is stored once, in its base form: a noun
# singular, a verb its bare stem, an adjective plain. Its regular forms come from these rules;
# only exceptions are stored, as forms ([w:goes->hasPastTense->w:went]), and a stored form
# blocks every rule. Closed-class words (is, has, does, ...) take stored forms only.
#
# A rule replaces the end of the base spelling (baseEnding; none = nothing) with formEnding.
# A letter-class wildcard (??.consonant) in both endings is the same letter, bound by reference
# as a template's roles are; where a class repeats, a role tells the letters apart.
#
# The rules form an is-a tree under each default rule. A child rule is an exception to its parent:
# it inherits what the parent forms (formsInflection; formsInflection.not overrides) and its
# backward preference (readsBackward, exclusive, so a child's own overrides). Its endings are its
# own. Forward, the deepest rule whose base ending the word has wins. Backward, every rule whose
# form ending the spelling has gives a candidate base; a candidate counts only if the rules run
# forward give the spelling back, and it is credited to the rule that did. Of a new word's
# candidates, a preferred rule's comes before an avoided one's, then a child rule's before its
# ancestor's. Preferences matter only for unknown words: a known base word is found regardless.
#
# Loaded after BasicWords.txt, which defines the letters, vowel/consonant and the inflections
# plural, pastTense, pastParticiple, presentParticiple.

[inflectionRule->is-a->languageElement]
[formsInflection->is-a->LinkType]
[baseEnding->is-a->LinkType]
[formEnding->is-a->LinkType]
[inflects->is-a->LinkType]
[lacksInflection->is-a->LinkType]
[letterRole->is-a->languageElement]
[letterRole->hasProperty->isWildcard]
[letterRole1->is-a->letterRole]
[letterRole2->is-a->letterRole]
[letterRole3->is-a->letterRole]
[readsBackward->is-a->LinkType]
[backwardPreference->is-a->languageElement]
[backwardPreference->hasProperty->isExclusive]
[preferred->is-a->backwardPreference]
[avoided->is-a->backwardPreference]
[inflectionRule->readsBackward->preferred]

# Exceptions are stored with the words (BasicWords.txt: went, gone, ate, was, were, are, ...).

# Derivations: an inflection that makes a word of another class (derives). It is applied to the
# class it inflects, as any inflection is, and spelled by rules the same way. A derived word
# whose meaning drifts (hardly, lately, shortly) is stored as a word with its own meaning; the
# base then stores its real adverb ([w:hard->hasAdverbial->...] spelled hard), which blocks the
# rule, so "hardly" is not read as "in a hard manner".
[derives->is-a->LinkType]
[derivation->is-a->inflection]
[adverb->is-a->wordType]
[mannerAdverb->is-a->adverb]
[adverbial->is-a->derivation]
[adverbial->inflects->adjective]
[adverbial->derives->mannerAdverb]
[hasAdverbial->is-a->hasInflection]
[hasAdverbial->marksInflection->adverbial]
